cmake_minimum_required(VERSION 3.24)
project(AegisDB VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(AEGISDB_BUILD_TESTS "Build GoogleTest and TCP integration tests" ON)
option(AEGISDB_BUILD_SERVER "Build the Boost.Beast server" ON)
set(AEGISDB_SANITIZER "" CACHE STRING "address or thread")
find_package(Threads REQUIRED)
if(AEGISDB_SANITIZER STREQUAL "address")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(AEGISDB_SANITIZER STREQUAL "thread")
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
add_library(aegisdb src/core/codec.cpp src/filesystem/posix.cpp src/storage/memtable.cpp
  src/storage/log.cpp src/storage/bloom.cpp src/storage/cache.cpp src/storage/sstable.cpp
  src/storage/manifest.cpp src/engine/db.cpp src/engine/db_impl.cpp src/engine/compaction.cpp)
target_include_directories(aegisdb PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE src)
target_link_libraries(aegisdb PUBLIC Threads::Threads)
target_compile_features(aegisdb PUBLIC cxx_std_20)
target_compile_options(aegisdb PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(aegisdb-example apps/embedded_example.cpp)
target_link_libraries(aegisdb-example PRIVATE aegisdb)
add_executable(aegisdb-benchmark apps/benchmark_main.cpp)
target_link_libraries(aegisdb-benchmark PRIVATE aegisdb)
if(AEGISDB_BUILD_SERVER)
  find_package(Boost 1.75 CONFIG REQUIRED COMPONENTS json)
  add_library(aegisdb_http src/server/server.cpp src/server/router.cpp src/server/config.cpp)
  target_include_directories(aegisdb_http PRIVATE src)
  target_link_libraries(aegisdb_http PUBLIC aegisdb Boost::headers Boost::json)
  target_compile_options(aegisdb_http PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(aegisdb-server apps/server_main.cpp)
  target_include_directories(aegisdb-server PRIVATE src)
  target_link_libraries(aegisdb-server PRIVATE aegisdb_http)
  target_compile_options(aegisdb-server PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS aegisdb EXPORT AegisDBTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/aegisdb DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(TARGETS aegisdb-example aegisdb-benchmark RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(AEGISDB_BUILD_SERVER)
  install(TARGETS aegisdb-server RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
configure_package_config_file(cmake/AegisDBConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/AegisDBConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/AegisDB)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/AegisDBConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT AegisDBTargets NAMESPACE AegisDB:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/AegisDB)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/AegisDBConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/AegisDBConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/AegisDB)
if(AEGISDB_BUILD_TESTS)
  enable_testing()
  find_package(GTest CONFIG REQUIRED)
  add_executable(aegisdb_tests tests/foundation_test.cpp tests/sstable_test.cpp tests/engine_test.cpp
    tests/recovery_test.cpp tests/base64_test.cpp)
  target_include_directories(aegisdb_tests PRIVATE src)
  target_link_libraries(aegisdb_tests PRIVATE aegisdb GTest::gtest_main)
  # Older externally packaged GoogleTest targets may omit their macOS @rpath directory.
  get_target_property(_gtest_location GTest::gtest IMPORTED_LOCATION_RELEASE)
  if(_gtest_location)
    get_filename_component(_gtest_directory "${_gtest_location}" DIRECTORY)
    set_property(TARGET aegisdb_tests APPEND PROPERTY BUILD_RPATH "${_gtest_directory}")
  endif()
  include(GoogleTest)
  gtest_discover_tests(aegisdb_tests DISCOVERY_TIMEOUT 30 PROPERTIES TIMEOUT 60)
  add_test(NAME Benchmark.Smoke COMMAND aegisdb-benchmark --count 128 --value-size 32)
  set_tests_properties(Benchmark.Smoke PROPERTIES TIMEOUT 30 LABELS "integration")
  add_custom_target(check-unit COMMAND ${CMAKE_CTEST_COMMAND} --output-on-failure -R "^(Codec|SkipList|Memtable|WAL|Bloom|SSTable|Base64)\\." DEPENDS aegisdb_tests WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR} VERBATIM)
  add_custom_target(check-integration COMMAND ${CMAKE_CTEST_COMMAND} --output-on-failure -R "^(EngineTest|Recovery|HttpRouter|Benchmark|Server)\\." DEPENDS aegisdb_tests aegisdb-benchmark WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR} VERBATIM)
  if(AEGISDB_BUILD_SERVER)
    add_dependencies(check-integration aegisdb-server)
    target_sources(aegisdb_tests PRIVATE tests/router_test.cpp)
    target_link_libraries(aegisdb_tests PRIVATE aegisdb_http)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME Server.RealTCP COMMAND Python3::Interpreter ${CMAKE_CURRENT_SOURCE_DIR}/tests/server_test.py $<TARGET_FILE:aegisdb-server>)
    set_tests_properties(Server.RealTCP PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77 LABELS "server;integration")
  endif()
endif()
